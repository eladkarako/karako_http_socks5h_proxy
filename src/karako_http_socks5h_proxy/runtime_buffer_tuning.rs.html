1//! Process-wide buffer tuning chosen at startup.
2//!
3//! The socket and tunnel layers read these values on every new socket, so
4//! they are published once before the listener starts and then treated as
5//! immutable. A `OnceLock` keeps that read lock-free.
6
7use std::sync::OnceLock;
8
9/// Startup buffer tuning shared by the socket and tunnel layers.
10///
11/// This is set once before the listener starts, so plain reads are
12/// effectively immutable for the lifetime of the process.
13#[derive(Clone, Copy, Debug)]
14pub(crate) struct Tuning {
15    /// Global `SO_RCVBUF` override in bytes, if the user set one.
16    pub socket_recv_buffer: Option<usize>,
17    /// Global `SO_SNDBUF` override in bytes, if the user set one.
18    pub socket_send_buffer: Option<usize>,
19    /// CONNECT copy buffer size in bytes, applied per direction.
20    pub tunnel_buffer_size: usize,
21}
22
23impl Default for Tuning {
24    fn default() -> Self {
25        Self {
26            socket_recv_buffer: None,
27            socket_send_buffer: None,
28            tunnel_buffer_size: 32 * 1024,
29        }
30    }
31}
32
33static TUNING: OnceLock<Tuning> = OnceLock::new();
34
35/// Publishes the tuning values. Only the first call has an effect.
36pub(crate) fn init(tuning: Tuning) {
37    let _ = TUNING.set(tuning);
38}
39
40/// Returns the published values, or the defaults if [`init`] has not run.
41pub(crate) fn get() -> Tuning {
42    TUNING.get().copied().unwrap_or_default()
43}
