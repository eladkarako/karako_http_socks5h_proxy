1//! Per-request handling for HTTP/2 clients.
2//!
3//! HTTP/2 runs over the same `hyper-util` auto server as HTTP/1.x; this
4//! module only differs in its log label. `CONNECT` uses the extended
5//! protocol enabled in the service builder, while ordinary requests take
6//! the same forwarding path as HTTP/1.1.
7
8use http::{Request, Response};
9use hyper::body::{Body, Incoming};
10
11use crate::proxy_http_forwarder;
12use crate::proxy_http_forwarder::{
13    build_forward_request, extract_target, BoxBody, ForwardingOptions,
14};
15use crate::socks5_handshake_flow::Socks5Auth;
16use crate::upstream_http_client::UpstreamClient;
17
18/// Handles one HTTP/2 stream.
19///
20/// Each stream is independent, which is why the service caps concurrent
21/// streams: every forwarded request opens its own upstream SOCKS5
22/// connection.
23pub(crate) async fn handle_h2_request(
24    req: Request<Incoming>,
25    upstream_addr: std::net::SocketAddr,
26    client_addr: std::net::SocketAddr,
27    client: UpstreamClient,
28    auth: Socks5Auth,
29    forwarding: ForwardingOptions,
30) -> anyhow::Result<Response<BoxBody>> {
31    let started = std::time::Instant::now();
32    let method = req.method().clone();
33    let uri = req.uri().clone();
34    let version = req.version();
35    let header_count = req.headers().len();
36    let target = extract_target(&req)?;
37    let (host, port) = (target.host, target.port);
38    log_info!(
39        "[HTTP/2] {} {} (host: {}:{}) from {}",
40        method,
41        proxy_http_forwarder::loggable_target(&uri),
42        host,
43        port,
44        client_addr
45    );
46    log_debug!(
47        "[HTTP/2] {} header(s), body size hint: {:?}",
48        header_count,
49        req.body().size_hint()
50    );
51    let response = match method.as_str() {
52        // HTTP/2 CONNECT carries `:authority` instead of a `Host`
53        // header; `extract_target` has already normalized both forms.
54        "CONNECT" => {
55            crate::http_connect_tunnel::connect_through_socks5(
56                req,
57                host,
58                port,
59                upstream_addr,
60                auth,
61                forwarding,
62            )
63                .await
64        }
65        _ => {
66            let forward_req = build_forward_request(
67                &method,
68                &uri,
69                &host,
70                port,
71                req,
72                client_addr,
73                forwarding,
74            )?;
75            proxy_http_forwarder::send_http_request(
76                &client,
77                forward_req,
78                version,
79                forwarding,
80            )
81                .await
82        }
83    };
84    match &response {
85        Ok(response) => log_debug!(
86            "[HTTP/2] {} {} -> {} in {:?}",
87            method,
88            proxy_http_forwarder::loggable_target(&uri),
89            response.status(),
90            started.elapsed()
91        ),
92        Err(error) => log_debug!(
93            "[HTTP/2] {} {} failed in {:?}: {:#}",
94            method,
95            proxy_http_forwarder::loggable_target(&uri),
96            started.elapsed(),
97            error
98        ),
99    }
100    response
101}
