1//! Connection-level HTTP service shared by both protocol versions.
2//!
3//! `hyper-util`'s auto server detects HTTP/1.1 versus HTTP/2 and calls
4//! [`ProxyService`] for each request or stream. The service enforces
5//! optional client authentication, dispatches to the version-specific
6//! handler, and converts any error into the status code carried by
7//! [`ProxyError`].
8

31/// How long a client may take to send request headers.
32///
33/// Slow-header attacks would otherwise hold a connection slot open
34/// indefinitely; the value is generous enough for slow links.
35const HEADER_READ_TIMEOUT: Duration = Duration::from_secs(30);
36
37/// Upper bound on concurrently multiplexed HTTP/2 streams per client
38/// connection. Each stream opens its own upstream SOCKS5 socket, so this
39/// keeps a single connection from exhausting upstream resources.
40const H2_MAX_CONCURRENT_STREAMS: u32 = 128;
41
42/// PING interval used to detect dead HTTP/2 peers.
43const H2_KEEP_ALIVE_INTERVAL: Duration = Duration::from_secs(30);
44
45/// How long to wait for a PING acknowledgement before closing the
46/// connection.
47const H2_KEEP_ALIVE_TIMEOUT: Duration = Duration::from_secs(20);

49/// Per-connection services and policies.
50///
51/// Grouping them keeps [`handle_client_connection`] small and lets the
52/// accept loop build the value once, then clone it per connection. The
53/// upstream client is a pool handle, so cloning is cheap and all clones
54/// share the same connections.
55#[derive(Clone)]
56pub struct ProxyServiceConfig {
57    /// SOCKS5 server every forwarded request is sent through.
58    pub upstream_addr: SocketAddr,
59    /// Pooled upstream HTTP client.
60    pub client: UpstreamClient,
61    /// Upstream SOCKS5 credentials.
62    pub auth: Socks5Auth,
63    /// Inbound client authentication policy.
64    pub inbound_auth: ProxyAuth,
65    /// Which optional headers to add.
66    pub forwarding: ForwardingOptions,
67    /// Which response rewrites to apply.
68    pub response_policy: ResponsePolicy,
69}

71/// Cloneable request handler for one accepted client connection.
72///
73/// The upstream client is a pool handle, so cloning is cheap and all
74/// clones share the same connections.
75#[derive(Clone)]
76struct ProxyService {
77    /// SOCKS5 server every forwarded request is sent through.
78    upstream_addr: SocketAddr,
79    /// Peer address, used for logs and `X-Forwarded-For`.
80    client_addr: SocketAddr,
81    /// Pooled upstream HTTP client.
82    client: UpstreamClient,
83    /// Upstream SOCKS5 credentials.
84    auth: Socks5Auth,
85    /// Inbound client authentication policy.
86    inbound_auth: ProxyAuth,
87    /// Which optional headers to add.
88    forwarding: ForwardingOptions,
89    /// Which response rewrites to apply.
90    response_policy: ResponsePolicy,
91}
92
93impl Service<Request<Incoming>> for ProxyService {
94    type Response = Response<BoxBody>;
95    type Error = Infallible;
96    type Future = Pin<
97        Box<
98            dyn Future<Output=Result<Self::Response, Self::Error>>
99            + Send
100            + 'static,
101        >,
102    >;
103
104    fn call(
105        &self,
106        req: Request<Incoming>,
107    ) -> Self::Future {
108        let upstream_addr = self.upstream_addr;
109        let client_addr = self.client_addr;
110        let client = self.client.clone();
111        let auth = self.auth.clone();
112        let inbound_auth = self.inbound_auth.clone();
113        let forwarding = self.forwarding;
114        let response_policy = self.response_policy;
115        Box::pin(async move {
116            // Cross-domain mocks only exist when permissive CORS is on.
117            // Resolving them before dispatch avoids cloning or moving the
118            // request twice.
119            let cross_domain = if response_policy.cors_allow_all {
120                response_header_policy::cross_domain_mock(
121                    req.method(),
122                    req.uri(),
123                )
124            } else {
125                None
126            };
127            let mut response = if !inbound_auth.is_authorized(&req) {
128                log_warn!(
129                    "[AUTH] Rejecting unauthenticated request from {}",
130                    client_addr
131                );
132                challenge()
133            } else if let Some(body) = cross_domain {
134                log_info!(
135                    "[CORS] Serving synthetic cross-domain policy for {}",
136                    req.uri().path()
137                );
138                response_header_policy::cross_domain_response(body)
139            } else {
140                let result = match req.version() {
141                    hyper::Version::HTTP_2 => {
142                        handle_h2_request(
143                            req,
144                            upstream_addr,
145                            client_addr,
146                            client,
147                            auth,
148                            forwarding,
149                        )
150                            .await
151                    }
152                    _ => {
153                        handle_http1_request(
154                            req,
155                            upstream_addr,
156                            client_addr,
157                            client,
158                            auth,
159                            forwarding,
160                        )
161                            .await
162                    }
163                };
164                result.unwrap_or_else(|e| {
165                    log_error!(
166                        "Request handling error from {}: {:#}",
167                        client_addr,
168                        e
169                    );
170                    error_response(&e)
171                })
172            };
173            response_policy.apply(&mut response);
174            Ok(response)
175        })
176    }
177}

179/// Maps a handler error to the response sent to the client.
180///
181/// `BadRequest` keeps its message because it describes a client mistake;
182/// upstream failures use fixed text so internal details stay in the logs.
183fn error_response(error: &anyhow::Error) -> Response<BoxBody> {
184    let (status, message) =
185        match error.downcast_ref::<ProxyError>() {
186            Some(ProxyError::BadRequest(message)) => {
187                (StatusCode::BAD_REQUEST, message.clone())
188            }
189            Some(ProxyError::Timeout { .. }) => (
190                StatusCode::GATEWAY_TIMEOUT,
191                "upstream request timed out".to_string(),
192            ),
193            None => (
194                StatusCode::BAD_GATEWAY,
195                "upstream request failed".to_string(),
196            ),
197        };
198    let body = full_body(message);
199    Response::builder()
200        .status(status)
201        .body(body)
202        .expect("valid error response")
203}

205/// Serves one client connection until it closes or errors.
206///
207/// Protocol detection and connection reuse are delegated to `hyper-util`.
208/// Returning `Ok(())` for a connection-level error is deliberate: the
209/// error is already logged, and the listener should keep running.
210pub async fn handle_client_connection(
211    socket: TcpStream,
212    addr: SocketAddr,
213    config: ProxyServiceConfig,
214) -> Result<()> {
215    let ProxyServiceConfig {
216        upstream_addr,
217        client,
218        auth,
219        inbound_auth,
220        forwarding,
221        response_policy,
222    } = config;
223    let started = std::time::Instant::now();
224    let io = hyper_util::rt::TokioIo::new(socket);
225    let service = ProxyService {
226        upstream_addr,
227        client_addr: addr,
228        client,
229        auth,
230        inbound_auth,
231        forwarding,
232        response_policy,
233    };
234    let mut builder = hyper_util::server::conn::auto::Builder::new(
235        hyper_util::rt::TokioExecutor::new(),
236    );
237    builder
238        .http1()
239        .timer(hyper_util::rt::TokioTimer::new())
240        .preserve_header_case(true)
241        .title_case_headers(false)
242        .header_read_timeout(HEADER_READ_TIMEOUT);
243    builder
244        .http2()
245        .timer(hyper_util::rt::TokioTimer::new())
246        .enable_connect_protocol()
247        .max_concurrent_streams(H2_MAX_CONCURRENT_STREAMS)
248        .keep_alive_interval(H2_KEEP_ALIVE_INTERVAL)
249        .keep_alive_timeout(H2_KEEP_ALIVE_TIMEOUT);
250    let result = builder
251        .serve_connection_with_upgrades(io, service)
252        .await;
253    log_debug!(
254        "[CLOSE] {} disconnected after {:?}",
255        addr,
256        started.elapsed()
257    );
258    match result {
259        Ok(()) => Ok(()),
260        Err(e) => {
261            log_warn!("Connection {} error: {}", addr, e);
262            Ok(())
263        }
264    }
265}
