1//! Command-line surface for the proxy.
2//!
3//! `clap` derives `--help` from the doc comments on [`Args`] and its
4//! fields: the first line of each comment becomes the short help shown
5//! by `-h`, and any following paragraph becomes the long help shown by
6//! `--help`. Keep those comments descriptive enough to stand alone.
7//!
8//! [`parse_and_validate`] is the only entry point the rest of the
9//! program uses; it resolves the SOCKS5 password sources and rejects
10//! configurations that would fail later anyway.
11
12use anyhow::{bail, ensure, Context, Result};
13use clap::Parser;
14use std::net::{IpAddr, SocketAddr};
15use std::path::PathBuf;
16
17/// Smallest accepted buffer size, in KiB.
18const MIN_BUFFER_KIB: u32 = 5;
19
20/// Largest accepted buffer size, in KiB (10 MiB).
21const MAX_BUFFER_KIB: u32 = 10 * 1024;
22
23/// Human-readable form of the accepted buffer range, following
24/// "must be between".
25const BUFFER_RANGE_HINT: &str = "5 and 10240 KiB (5 KiB to 10 MiB)";
26
27/// A forward proxy that carries HTTP traffic over an upstream SOCKS5h
28/// server.
29///
30/// Clients speak HTTP/1.1 or HTTP/2 to this process. Plain HTTP requests
31/// are forwarded as requests, while `CONNECT` opens a raw tunnel for TLS
32/// and anything else. Destination names are handed to the SOCKS5 server,
33/// so DNS resolution happens on the far side of the tunnel.
34#[derive(Parser, Debug, Clone)]
35#[command(name = env!("CARGO_PKG_NAME"))]
36#[command(version = env!("CARGO_PKG_VERSION"))]
37#[command(author = env!("CARGO_PKG_AUTHORS"))]
38#[command(
39    about = env!("CARGO_PKG_DESCRIPTION"),
40    long_about = concat!(
41        "HTTP-over-SOCKS5h forward proxy.\n\n",
42        "Listens for HTTP/1.1 and HTTP/2 clients and forwards their ",
43        "requests through an upstream SOCKS5h server. Host names travel ",
44        "to that server unresolved, so remote DNS is used.\n\n",
45        "Plain HTTP is forwarded directly; HTTPS and other TLS traffic ",
46        "must use CONNECT.\n\n",
47        "Example:\n",
48        "  karako_http_socks5h_proxy -p 18765 -u 9150\n",
49        "  curl -x http://127.0.0.1:18765 http://example.com/"
50    )
51)]
52#[command(
53    after_help = concat!("Repository:    ", env!("CARGO_PKG_REPOSITORY"))
54)]
55pub struct Args {
56    /// Port the HTTP proxy listens on for clients
57    ///
58    /// Point clients at `http://<bind>:<http-port>`. This is the port a
59    /// browser or `curl -x` should use, not the SOCKS5 port.
60    #[arg(short = 'p', long, default_value = "8765")]
61    pub http_port: u16,
62
63    /// Port of the upstream SOCKS5 server
64    ///
65    /// Usually a local Tor instance (9050, or 9150 for the Tor Browser
66    /// bundle). The proxy never speaks SOCKS5 to clients.
67    #[arg(short = 'u', long, default_value = "5678")]
68    pub upstream_port: u16,
69
70    /// Address to bind the HTTP proxy listener on
71    ///
72    /// Defaults to loopback. Any non-loopback address requires
73    /// `--allow-non-local`, because the listener exposes the upstream
74    /// SOCKS5 server to everyone who can reach it.
75    #[arg(short = 'b', long, default_value = "127.0.0.1")]
76    pub bind: IpAddr,
77
78    /// Address of the upstream SOCKS5 server
79    ///
80    /// Separate from `--bind` so the proxy can listen on one interface
81    /// while reaching the SOCKS5 server over another.
82    #[arg(long, default_value = "127.0.0.1")]
83    pub upstream_host: IpAddr,
84
85    /// Username for SOCKS5 username/password authentication
86    ///
87    /// Requires a password from `--socks5-password`,
88    /// `--socks5-password-file`, or `KARAKO_SOCKS5_PASSWORD`. When this
89    /// is omitted, only the no-authentication method is offered.
90    #[arg(long)]
91    pub socks5_username: Option<String>,
92
93    /// Password for SOCKS5 username/password authentication
94    ///
95    /// Prefer `--socks5-password-file` or `KARAKO_SOCKS5_PASSWORD`:
96    /// command-line arguments are visible to other users through the
97    /// process list.
98    #[arg(long, conflicts_with = "socks5_password_file")]
99    pub socks5_password: Option<String>,
100
101    /// File containing the SOCKS5 password
102    ///
103    /// One trailing newline (LF or CRLF) is stripped. Takes precedence
104    /// over `KARAKO_SOCKS5_PASSWORD`, which takes precedence over
105    /// `--socks5-password`.
106    #[arg(long, value_name = "PATH")]
107    pub socks5_password_file: Option<PathBuf>,
108
109    /// Username HTTP clients must send via Proxy-Authorization
110    ///
111    /// Turns the listener into an authenticated proxy. Requires
112    /// `--proxy-password`. Clients then use
113    /// `Proxy-Authorization: Basic <base64(user:pass)>`, which `curl`
114    /// does automatically for `-x http://user:pass@host:port`.
115    #[arg(long, requires = "proxy_password")]
116    pub proxy_username: Option<String>,
117
118    /// Password HTTP clients must send via Proxy-Authorization
119    ///
120    /// Requests without a matching header receive `407 Proxy
121    /// Authentication Required` with a Basic challenge.
122    #[arg(long, requires = "proxy_username")]
123    pub proxy_password: Option<String>,
124
125    /// Allow binding to a non-loopback address
126    ///
127    /// Without this flag the proxy refuses to start on a public
128    /// interface. Anyone who can reach the listener can use the upstream
129    /// SOCKS5 server, so pair this with client authentication.
130    #[arg(long)]
131    pub allow_non_local: bool,
132
133    /// Omit the Via header from forwarded messages
134    ///
135    /// `Via` identifies this proxy in requests and responses (RFC 9110).
136    /// Use this when the proxy should stay invisible to the origin
137    /// server.
138    #[arg(long)]
139    pub no_via: bool,
140
141    /// Add X-Forwarded-For with the client address
142    ///
143    /// A client-supplied `X-Forwarded-For` is discarded and replaced, so
144    /// the header cannot be used to spoof the original address.
145    #[arg(long)]
146    pub x_forwarded_for: bool,
147
148    /// Extend Cache-Control with stale-while-revalidate and stale-if-error
149    ///
150    /// Removes any existing `stale-while-revalidate`/`stale-if-error`
151    /// directives, then appends `stale-while-revalidate=86400` (one day)
152    /// and `stale-if-error=259200` (three days) to every response. A
153    /// homage to the historic "Boostaler" browser extension, which aimed
154    /// to keep pages usable while a site was slow or down.
155    #[arg(long, visible_alias = "stale-cache")]
156    pub boostaler_cache: bool,
157
158    /// Add permissive CORS headers to every response
159    ///
160    /// Replaces existing values for Access-Control-Allow-Methods,
161    /// -Origin, -Headers, -Allow-Credentials, Expose-Headers,
162    /// Timing-Allow-Origin and X-Permitted-Cross-Domain-Policies, on
163    /// every method including non-OPTIONS. Any path ending in
164    /// `CrossDomain.xml` or `ClientAccessPolicy.xml` (any letter case)
165    /// receives a synthetic 200 policy file instead of the upstream
166    /// response. CONNECT tunnels carry opaque TLS after the proxy's
167    /// 200, so headers inside the tunnel cannot be changed.
168    #[arg(long, visible_alias = "cors")]
169    pub cors_allow_all: bool,
170
171    /// Remove CSP and related security headers from every response
172    ///
173    /// Strips P3P, Alt-Svc, Strict-Transport-Security, X-Frame-Options,
174    /// X-XSS-Protection, X-Content-Type-Options, Content-Security-Policy,
175    /// Referrer-Policy and Public-Key-Pins-Report-Only, matching names in
176    /// any letter case.
177    #[arg(long, visible_alias = "strip-csp")]
178    pub strip_security_headers: bool,
179
180    /// Copy buffer size per direction for CONNECT tunnels, in KiB
181    ///
182    /// Each direction of a tunnel gets its own buffer of this size. The
183    /// default suits most links; raise it for high-bandwidth tunnels on
184    /// high-latency paths. Accepted range: 5 to 10240 KiB (5 KiB to
185    /// 10 MiB).
186    #[arg(long, value_name = "KIB", default_value_t = 32)]
187    pub tunnel_buffer_size: u16,
188
189    /// Override SO_RCVBUF for all sockets, in KiB (default: 64-512 by role)
190    ///
191    /// Applies to the listener, SOCKS5 handshakes, upstream HTTP
192    /// connections, and tunnels. Without this flag each role keeps its
193    /// own default: 64 KiB for handshakes, 256 KiB for inbound and HTTP
194    /// request sockets, and 512 KiB for CONNECT tunnels. Accepted range:
195    /// 5 to 10240 KiB (5 KiB to 10 MiB).
196    #[arg(long, value_name = "KIB")]
197    pub socket_recv_buffer: Option<u32>,
198
199    /// Override SO_SNDBUF for all sockets, in KiB (default: 64-512 by role)
200    ///
201    /// Same coverage and defaults as `--socket-recv-buffer`: 64 KiB for
202    /// handshakes, 256 KiB for inbound and HTTP request sockets, and
203    /// 512 KiB for CONNECT tunnels. Accepted range: 5 to 10240 KiB
204    /// (5 KiB to 10 MiB). The kernel may clamp the requested value; the
205    /// effective size is logged with `-v`.
206    #[arg(long, value_name = "KIB")]
207    pub socket_send_buffer: Option<u32>,
208
209    /// Increase log verbosity
210    ///
211    /// No flag logs one line per request. `-v` adds handshake, socket and
212    /// timing details. `-vv` also dumps SOCKS5 packets as hex.
213    #[arg(short = 'v', long, action = clap::ArgAction::Count)]
214    pub verbose: u8,
215
216    /// Only log errors
217    ///
218    /// Mutually exclusive with `--verbose`. Useful when the proxy runs in
219    /// the background and only failures are interesting.
220    #[arg(short = 'q', long, conflicts_with = "verbose")]
221    pub quiet: bool,
222}

224/// Parses the command line, resolves the SOCKS5 password, and checks the
225/// resulting configuration.
226///
227/// Validation happens here rather than at first use so that a bad
228/// combination fails before the listener is bound.
229pub fn parse_and_validate() -> Result<Args> {
230    let mut args = Args::parse();
231    resolve_socks5_password(&mut args)?;
232    let listen = SocketAddr::new(args.bind, args.http_port);
233    let upstream =
234        SocketAddr::new(args.upstream_host, args.upstream_port);
235    ensure!(
236        listen != upstream,
237        "HTTP listener ({}) cannot be the same as the upstream \
238         SOCKS5 server ({})",
239        listen,
240        upstream
241    );
242    ensure!(
243        args.bind.is_loopback() || args.allow_non_local,
244        "refusing to bind the proxy to non-loopback address {}; \
245         pass --allow-non-local to expose it to the network",
246        args.bind
247    );
248    ensure!(
249        (MIN_BUFFER_KIB..=MAX_BUFFER_KIB)
250            .contains(&u32::from(args.tunnel_buffer_size)),
251        "--tunnel-buffer-size must be between {}",
252        BUFFER_RANGE_HINT
253    );
254    if let Some(size) = args.socket_recv_buffer {
255        ensure!(
256            (MIN_BUFFER_KIB..=MAX_BUFFER_KIB).contains(&size),
257            "--socket-recv-buffer must be between {}",
258            BUFFER_RANGE_HINT
259        );
260    }
261    if let Some(size) = args.socket_send_buffer {
262        ensure!(
263            (MIN_BUFFER_KIB..=MAX_BUFFER_KIB).contains(&size),
264            "--socket-send-buffer must be between {}",
265            BUFFER_RANGE_HINT
266        );
267    }
268    Ok(args)
269}

271/// Converts the KiB command-line values into bytes for the rest of the
272/// program.
273impl Args {
274    /// Tunnel copy buffer in bytes, used per direction.
275    pub fn tunnel_buffer_bytes(&self) -> usize {
276        usize::from(self.tunnel_buffer_size) * 1024
277    }
278
279    /// Global receive-buffer override in bytes, if one was requested.
280    pub fn socket_recv_buffer_bytes(&self) -> Option<usize> {
281        self.socket_recv_buffer
282            .map(|kib| kib as usize * 1024)
283    }
284
285    /// Global send-buffer override in bytes, if one was requested.
286    pub fn socket_send_buffer_bytes(&self) -> Option<usize> {
287        self.socket_send_buffer
288            .map(|kib| kib as usize * 1024)
289    }
290}

292/// Resolves the SOCKS5 password from the file, the environment, or the
293/// command line, in that order, and validates that a username is paired
294/// with a password.
295///
296/// Keeping this separate from clap's `requires` lets one password source
297/// satisfy `--socks5-username` regardless of which mechanism supplied it.
298fn resolve_socks5_password(args: &mut Args) -> Result<()> {
299    if let Some(path) = args.socks5_password_file.as_deref() {
300        let contents =
301            std::fs::read_to_string(path).with_context(|| {
302                format!(
303                    "Failed to read SOCKS5 password file {}",
304                    path.display()
305                )
306            })?;
307        // Editors on Windows often leave a CRLF behind; strip either
308        // ending so the password matches what the user typed.
309        args.socks5_password = Some(
310            contents
311                .trim_end_matches(|c| c == '\r' || c == '\n')
312                .to_string(),
313        );
314    }
315    if args.socks5_password.is_none() {
316        if let Ok(password) = std::env::var("KARAKO_SOCKS5_PASSWORD") {
317            args.socks5_password = Some(password);
318        }
319    }
320    match (
321        args.socks5_username.is_some(),
322        args.socks5_password.is_some(),
323    ) {
324        (true, false) => bail!(
325            "--socks5-username requires --socks5-password, \
326             --socks5-password-file, or KARAKO_SOCKS5_PASSWORD"
327        ),
328        (false, true) => bail!(
329            "a SOCKS5 password was provided without --socks5-username"
330        ),
331        _ => {}
332    }
333    Ok(())
