1//! SOCKS5 client handshake used for every upstream connection.
2//!
3//! The proxy never accepts SOCKS5; it only speaks the client half of
4//! RFC 1928 to reach the upstream server, plus RFC 1929
5//! username/password authentication when credentials are configured.
6//!
7//! A session has three steps: greet with the supported authentication
8//! methods, authenticate if the server chose that method, then send a
9//! `CONNECT` for the destination and consume the bound-address reply.
10//! Host names are sent as-is so that the server resolves them remotely
11//! (the `s` in SOCKS5h).
12//!
13//! [`connect_to_socks5_server`] performs the TCP connect and
14//! [`establish_socks5_connection`] drives the protocol exchange, both
15//! under [`SOCKS5_HANDSHAKE_TIMEOUT`].
16

28// Protocol constants from RFC 1928 (SOCKS5) and RFC 1929
29// (username/password authentication).
30const SOCKS5_VERSION: u8 = 5;
31const SOCKS5_CMD_CONNECT: u8 = 1;
32const SOCKS5_AUTH_METHOD_NONE: u8 = 0;
33const SOCKS5_AUTH_METHOD_USER_PASS: u8 = 0x02;
34const SOCKS5_AUTH_METHOD_UNACCEPTABLE: u8 = 0xFF;
35const SOCKS5_AUTH_VERSION: u8 = 0x01;
36const SOCKS5_ATYP_DOMAIN: u8 = 3;
37const SOCKS5_ATYP_IPV4: u8 = 1;
38const SOCKS5_ATYP_IPV6: u8 = 4;
39const MAX_HOSTNAME_LEN: usize = 255;
40
41/// The whole handshake, not just TCP connect, must finish in this window.
42const SOCKS5_HANDSHAKE_TIMEOUT: Duration = Duration::from_secs(10);
43
44/// Human-readable forms of the `REP` field, indexed by its value.
45const SOCKS5_ERRORS: &[&str] = &[
46    "",
47    "General SOCKS server failure",
48    "Connection not allowed by ruleset",
49    "Network unreachable",
50    "Host unreachable",
51    "Connection refused",
52    "TTL expired",
53    "Command not supported",
54    "Address type not supported",
55];
56
57/// Optional RFC 1929 username/password credentials for the upstream
58/// SOCKS5 server.
59#[derive(Clone, Default)]
60pub struct Socks5Auth {
61    credentials: Option<(String, String)>,
62}
63
64impl Socks5Auth {
65    /// Builds credentials from the CLI options. Both values must be
66    /// present (the argument parser enforces that) or neither is used.
67    pub fn new(
68        username: Option<String>,
69        password: Option<String>,
70    ) -> Self {
71        match (username, password) {
72            (Some(username), Some(password)) => Self {
73                credentials: Some((username, password)),
74            },
75            _ => Self { credentials: None },
76        }
77    }
78}

80/// Opens a TCP connection to the SOCKS5 server and applies handshake
81/// socket settings. The protocol exchange is not started here.
82pub async fn connect_to_socks5_server(
83    upstream_addr: std::net::SocketAddr
84) -> Result<TcpStream> {
85    let started = std::time::Instant::now();
86    let addr = upstream_addr;
87    let stream = match timeout(
88        SOCKS5_HANDSHAKE_TIMEOUT,
89        TcpStream::connect(&addr),
90    )
91        .await
92    {
93        Ok(result) => result.with_context(|| {
94            format!(
95                "Failed to connect to SOCKS5 server at {}",
96                addr
97            )
98        })?,
99        Err(_) => {
100            return Err(ProxyError::Timeout {
101                phase: "connecting to the SOCKS5 server",
102                duration: SOCKS5_HANDSHAKE_TIMEOUT,
103            }
104                .into());
105        }
106    };
107    configure_tokio_socket(
108        &stream,
109        SocketConfiguration::SOCKS_HANDSHAKE,
110    );
111    log_debug!(
112        "[SOCKS5] Connected to {} in {:?}",
113        addr,
114        started.elapsed()
115    );
116    Ok(stream)
117}

119/// Runs the SOCKS5 greeting, optional authentication and `CONNECT`
120/// exchange on an already-connected stream.
121///
122/// The caller must not use the stream for anything else until this
123/// returns `Ok`, because the reply to `CONNECT` marks the point where
124/// raw tunnel bytes begin.
125pub async fn establish_socks5_connection(
126    stream: &mut TcpStream,
127    host: &str,
128    port: u16,
129    auth: &Socks5Auth,
130) -> Result<()> {
131    let started = std::time::Instant::now();
132    match timeout(
133        SOCKS5_HANDSHAKE_TIMEOUT,
134        perform_socks5_handshake(stream, host, port, auth),
135    )
136        .await
137    {
138        Ok(result) => result?,
139        Err(_) => {
140            return Err(ProxyError::Timeout {
141                phase: "SOCKS5 handshake",
142                duration: SOCKS5_HANDSHAKE_TIMEOUT,
143            }
144                .into());
145        }
146    }
147    log_info!(
148        "[SOCKS5] Handshake complete for {}:{} in {:?}",
149        host,
150        port,
151        started.elapsed()
152    );
153    Ok(())
154}

156/// Drives the greeting and the CONNECT exchange in order, stopping at
157/// the first rejected step.
158async fn perform_socks5_handshake(
159    stream: &mut TcpStream,
160    host: &str,
161    port: u16,
162    auth: &Socks5Auth,
163) -> Result<()> {
164    send_socks5_greeting(stream, auth).await?;
165    let method = receive_socks5_greeting_response(stream).await?;
166    match method {
167        SOCKS5_AUTH_METHOD_NONE => {}
168        SOCKS5_AUTH_METHOD_USER_PASS => {
169            perform_user_pass_auth(stream, auth).await?;
170        }
171        other => bail!(
172            "SOCKS5 server selected unsupported authentication \
173             method 0x{:02X}",
174            other
175        ),
176    }
177    send_socks5_connect_request(stream, host, port).await?;
178    receive_socks5_connect_response(stream).await
179}
180
181/// Offers the authentication methods this proxy can handle.
182async fn send_socks5_greeting(
183    stream: &mut TcpStream,
184    auth: &Socks5Auth,
185) -> Result<()> {
186    let methods: &[u8] = if auth.credentials.is_some() {
187        // Advertise none first: servers that accept it can skip the
188        // RFC 1929 exchange entirely.
189        &[SOCKS5_AUTH_METHOD_NONE, SOCKS5_AUTH_METHOD_USER_PASS]
190    } else {
191        &[SOCKS5_AUTH_METHOD_NONE]
192    };
193    let mut greeting = Vec::with_capacity(2 + methods.len());
194    greeting.push(SOCKS5_VERSION);
195    greeting.push(methods.len() as u8);
196    greeting.extend_from_slice(methods);
197    stream
198        .write_all(&greeting)
199        .await
200        .context("Failed to send SOCKS5 greeting")?;
201    stream
202        .flush()
203        .await
204        .context("Failed to flush SOCKS5 greeting")?;
205    log_trace!(
206        "[SOCKS5] Greeting out ({} bytes):\n{}",
207        greeting.len(),
208        crate::logging_output_setup::hex_dump(&greeting)
209    );
210    Ok(())
211}

213/// Reads the method the server selected and rejects `0xFF`.
214async fn receive_socks5_greeting_response(
215    stream: &mut TcpStream
216) -> Result<u8> {
217    let mut resp = [0u8; 2];
218    stream
219        .read_exact(&mut resp)
220        .await
221        .context("Failed to read SOCKS5 greeting response")?;
222    log_trace!(
223        "[SOCKS5] Greeting in ({} bytes):\n{}",
224        resp.len(),
225        crate::logging_output_setup::hex_dump(&resp)
226    );
227    ensure!(
228        resp[0] == SOCKS5_VERSION,
229        "Invalid SOCKS5 version: {} (expected {})",
230        resp[0],
231        SOCKS5_VERSION
232    );
233    ensure!(
234        resp[1] != SOCKS5_AUTH_METHOD_UNACCEPTABLE,
235        "SOCKS5 server rejected all offered authentication methods"
236    );
237    Ok(resp[1])
238}

240/// Sends RFC 1929 credentials and checks the one-byte status reply.
241async fn perform_user_pass_auth(
242    stream: &mut TcpStream,
243    auth: &Socks5Auth,
244) -> Result<()> {
245    let Some((username, password)) = auth.credentials.as_ref()
246    else {
247        bail!(
248            "SOCKS5 server requires username/password \
249             authentication, but no credentials were configured"
250        );
251    };
252    ensure!(
253        username.len() <= 255,
254        "SOCKS5 username is too long: {} bytes (max 255)",
255        username.len()
256    );
257    ensure!(
258        password.len() <= 255,
259        "SOCKS5 password is too long: {} bytes (max 255)",
260        password.len()
261    );
262    let mut request =
263        Vec::with_capacity(3 + username.len() + password.len());
264    request.push(SOCKS5_AUTH_VERSION);
265    request.push(username.len() as u8);
266    request.extend_from_slice(username.as_bytes());
267    request.push(password.len() as u8);
268    request.extend_from_slice(password.as_bytes());
269    stream
270        .write_all(&request)
271        .await
272        .context("Failed to send SOCKS5 username/password auth")?;
273    stream
274        .flush()
275        .await
276        .context("Failed to flush SOCKS5 username/password auth")?;
277    log_debug!(
278        "[SOCKS5] Username/password auth sent (username {} bytes, password {} bytes)",
279        username.len(),
280        password.len()
281    );
282    let mut resp = [0u8; 2];
283    stream
284        .read_exact(&mut resp)
285        .await
286        .context("Failed to read SOCKS5 auth response")?;
287    ensure!(
288        resp[0] == SOCKS5_AUTH_VERSION,
289        "Invalid SOCKS5 auth version in response: {} (expected {})",
290        resp[0],
291        SOCKS5_AUTH_VERSION
292    );
293    ensure!(
294        resp[1] == 0,
295        "SOCKS5 username/password authentication failed \
296         (status 0x{:02X})",
297        resp[1]
298    );
299    log_debug!("[SOCKS5] Username/password authentication accepted");
300    Ok(())
301}

303/// Destination encoded for the CONNECT request.
304///
305/// IP literals are sent as IP address types; everything else is sent as a
306/// domain name so that the SOCKS5 server does the DNS lookup.
307enum Socks5Address<'a> {
308    Ipv4([u8; 4]),
309    Ipv6([u8; 16]),
310    Domain(&'a str),
311}
312
313/// Classifies a host as IPv4, IPv6 or a domain name.
314///
315/// Brackets around IPv6 literals are removed first so that the address
316/// parses; the brackets matter in URLs but not on the wire.
317fn classify_host(host: &str) -> Result<Socks5Address<'_>> {
318    let bare = host
319        .strip_prefix('[')
320        .and_then(|h| h.strip_suffix(']'))
321        .unwrap_or(host);
322    if let Ok(ip) = bare.parse::<std::net::Ipv4Addr>() {
323        return Ok(Socks5Address::Ipv4(ip.octets()));
324    }
325    if let Ok(ip) = bare.parse::<std::net::Ipv6Addr>() {
326        return Ok(Socks5Address::Ipv6(ip.octets()));
327    }
328    ensure!(
329        !host.is_empty() && host.len() <= MAX_HOSTNAME_LEN,
330        "Invalid hostname length: {} bytes (expected 1..={})",
331        host.len(),
332        MAX_HOSTNAME_LEN
333    );
334    Ok(Socks5Address::Domain(host))
335}

337/// Encodes and sends the `CONNECT` request for `host:port`.
338async fn send_socks5_connect_request(
339    stream: &mut TcpStream,
340    host: &str,
341    port: u16,
342) -> Result<()> {
343    let address = classify_host(host)?;
344    let atyp = match address {
345        Socks5Address::Ipv4(_) => SOCKS5_ATYP_IPV4,
346        Socks5Address::Ipv6(_) => SOCKS5_ATYP_IPV6,
347        Socks5Address::Domain(_) => SOCKS5_ATYP_DOMAIN,
348    };
349    let address_len = match address {
350        Socks5Address::Ipv4(_) => 4,
351        Socks5Address::Ipv6(_) => 16,
352        Socks5Address::Domain(name) => 1 + name.len(),
353    };
354    let mut connect_req = Vec::with_capacity(4 + address_len + 2);
355    connect_req.extend_from_slice(&[
356        SOCKS5_VERSION,
357        SOCKS5_CMD_CONNECT,
358        0,
359        atyp,
360    ]);
361    match &address {
362        Socks5Address::Ipv4(octets) => {
363            connect_req.extend_from_slice(octets);
364        }
365        Socks5Address::Ipv6(octets) => {
366            connect_req.extend_from_slice(octets);
367        }
368        Socks5Address::Domain(name) => {
369            connect_req.push(name.len() as u8);
370            connect_req.extend_from_slice(name.as_bytes());
371        }
372    }
373    connect_req.extend_from_slice(&port.to_be_bytes());
374    log_debug!(
375        "[SOCKS5] CONNECT {}:{} (atyp=0x{:02X}, {} bytes)",
376        host,
377        port,
378        atyp,
379        connect_req.len()
380    );
381    log_trace!(
382        "[SOCKS5] CONNECT out ({} bytes):\n{}",
383        connect_req.len(),
384        crate::logging_output_setup::hex_dump(&connect_req)
385    );
386    stream
387        .write_all(&connect_req)
388        .await
389        .context("Failed to send SOCKS5 CONNECT")?;
390    stream
391        .flush()
392        .await
393        .context("Failed to flush SOCKS5 CONNECT")
394}

396/// Validates the `CONNECT` reply and consumes the bound address that
397/// follows it.
398async fn receive_socks5_connect_response(
399    stream: &mut TcpStream
400) -> Result<()> {
401    let mut resp_header = [0u8; 4];
402    stream
403        .read_exact(&mut resp_header)
404        .await
405        .context("Failed to read SOCKS5 CONNECT response")?;
406    log_debug!(
407        "[SOCKS5] CONNECT response: ver={}, rep={}, rsv={}, atyp={}",
408        resp_header[0],
409        resp_header[1],
410        resp_header[2],
411        resp_header[3]
412    );
413    log_trace!(
414        "[SOCKS5] CONNECT response header ({} bytes):\n{}",
415        resp_header.len(),
416        crate::logging_output_setup::hex_dump(&resp_header)
417    );
418    ensure!(
419        resp_header[0] == SOCKS5_VERSION,
420        "Invalid SOCKS5 version in response: {} (expected {})",
421        resp_header[0],
422        SOCKS5_VERSION
423    );
424    ensure!(
425        resp_header[2] == 0,
426        "Invalid SOCKS5 reserved byte in response: 0x{:02X}",
427        resp_header[2]
428    );
429    if resp_header[1] != 0 {
430        let error_msg =
431            if (resp_header[1] as usize) < SOCKS5_ERRORS.len() {
432                SOCKS5_ERRORS[resp_header[1] as usize].to_string()
433            } else {
434                format!("Unknown error code {}", resp_header[1])
435            };
436        bail!("SOCKS5 CONNECT failed: {}", error_msg);
437    }
438    read_socks5_bind_address(stream, resp_header[3]).await
439}

441/// Discards the address the server bound on its side.
442///
443/// The proxy never uses this address; it only has to read the bytes so
444/// that the next byte on the stream is the start of tunnel data.
445async fn read_socks5_bind_address(
446    stream: &mut TcpStream,
447    atyp: u8,
448) -> Result<()> {
449    match atyp {
450        SOCKS5_ATYP_IPV4 => {
451            let mut addr_port = [0u8; 6];
452            stream
453                .read_exact(&mut addr_port)
454                .await
455                .context("Failed to read SOCKS5 IPv4 response")?;
456        }
457        SOCKS5_ATYP_DOMAIN => {
458            let mut len_byte = [0u8; 1];
459            stream
460                .read_exact(&mut len_byte)
461                .await
462                .context("Failed to read SOCKS5 domain length")?;
463            let domain_len = len_byte[0] as usize;
464            ensure!(
465                domain_len > 0 && domain_len <= MAX_HOSTNAME_LEN,
466                "Invalid SOCKS5 domain length: {}",
467                domain_len
468            );
469            let mut domain_and_port = vec![0u8; domain_len + 2];
470            stream
471                .read_exact(&mut domain_and_port)
472                .await
473                .context("Failed to read SOCKS5 domain response")?;
474        }
475        SOCKS5_ATYP_IPV6 => {
476            let mut addr_port = [0u8; 18];
477            stream
478                .read_exact(&mut addr_port)
479                .await
480                .context("Failed to read SOCKS5 IPv6 response")?;
481        }
482        atype => {
483            bail!(
484                "Unsupported SOCKS5 address type in response: {}",
485                atype
486            );
487        }
488    }
489    log_trace!(
490        "[SOCKS5] Bind address consumed (atyp=0x{:02X})",
491        atyp
492    );
493    Ok(())
