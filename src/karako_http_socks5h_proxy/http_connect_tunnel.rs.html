1//! `CONNECT` handling and the raw byte tunnel behind it.
2//!
3//! A successful `CONNECT` has two parts. First the proxy opens a SOCKS5
4//! connection to the requested authority and answers `200 OK`; `hyper`
5//! then upgrades the client connection and hands it back. Second, the
6//! upgraded stream and the SOCKS5 socket are spliced together and copied
7//! in both directions until one side closes or the idle timeout fires.
8
9use anyhow::{Context, Result};
10use http::{Request, Response, StatusCode};
11use hyper::body::Incoming;
12use std::time::Duration;
13use tokio::net::TcpStream;
14use tokio_io_timeout::TimeoutStream;
15
16use crate::proxy_http_forwarder::{
17    empty_body, BoxBody, ForwardingOptions,
18};
19use crate::tcp_socket_config::{
20    configure_tokio_socket, SocketConfiguration,
21};
22use crate::socks5_handshake_flow::{
23    connect_to_socks5_server, establish_socks5_connection,
24    Socks5Auth,
25};
26
27/// How long a tunnel may sit without any traffic in either direction
28/// before it is torn down.
29pub(crate) const TUNNEL_IDLE_TIMEOUT: Duration = Duration::from_secs(300);
30
31/// Handles a `CONNECT` request by opening a SOCKS5 tunnel to the
32/// requested authority and upgrading the client connection to a raw
33/// byte stream.
34pub async fn connect_through_socks5(
35    req: Request<Incoming>,
36    host: String,
37    port: u16,
38    upstream_addr: std::net::SocketAddr,
39    auth: Socks5Auth,
40    forwarding: ForwardingOptions,
41) -> Result<Response<BoxBody>> {
42    let started = std::time::Instant::now();
43    let client_version = req.version();
44    log_info!(
45        "[CONNECT TUNNEL] Initiating tunnel to {}:{}",
46        host,
47        port
48    );
49    let mut target_stream =
50        connect_to_socks5_server(upstream_addr).await?;
51    establish_socks5_connection(
52        &mut target_stream,
53        &host,
54        port,
55        &auth,
56    )
57        .await?;
58    configure_tokio_socket(
59        &target_stream,
60        SocketConfiguration::SOCKS_TUNNEL,
61    );
62    log_debug!(
63        "[CONNECT TUNNEL] SOCKS5 connection established in {:?}, sending 200 OK",
64        started.elapsed()
65    );
66    // A 2xx response to CONNECT switches both HTTP/1.1 and HTTP/2 to a
67    // raw tunnel, so no upgrade headers are needed (and hop-by-hop
68    // headers must not be sent at all).
69    let mut response_builder =
70        Response::builder().status(StatusCode::OK);
71    if forwarding.add_via {
72        response_builder = response_builder.header(
73            http::header::VIA,
74            crate::proxy_http_forwarder::via_value(client_version),
75        );
76    }
77    let response = response_builder
78        .body(empty_body())
79        .context("Failed to build CONNECT response")?;
80    // The upgrade future must be polled after the 200 response is
81    // returned, so the copy runs in its own task. Moving both sockets
82    // into the task also keeps the client connection handler free.
83    tokio::spawn(async move {
84        match hyper::upgrade::on(req).await {
85            Ok(upgraded) => {
86                log_debug!(
87                    "[CONNECT TUNNEL] Upgrade successful for {}:{}",
88                    host,
89                    port
90                );
91                if let Err(e) = handle_tunnel(
92                    upgraded,
93                    target_stream,
94                    &host,
95                    port,
96                )
97                    .await
98                {
99                    log_error!(
100                        "[CONNECT TUNNEL] Tunnel error for {}:{}: {:#}",
101                        host,
102                        port,
103                        e
104                    );
105                }
106            }
107            Err(e) => {
108                log_warn!(
109                    "[CONNECT TUNNEL] Upgrade failed for {}:{}: {}",
110                    host,
111                    port,
112                    e
113                );
114            }
115        }
116    });
117    Ok(response)
118}

120/// Copies bytes in both directions until either side closes or the idle
121/// timeout expires, then logs the byte counts.
122///
123/// `TimeoutStream` applies the idle timeout per read and per write, so a
124/// tunnel that keeps moving data stays open indefinitely.
125pub async fn handle_tunnel(
126    upgraded: hyper::upgrade::Upgraded,
127    target: TcpStream,
128    host: &str,
129    port: u16,
130) -> Result<()> {
131    let started = std::time::Instant::now();
132    log_debug!(
133        "[TUNNEL] Starting bidirectional tunnel for {}:{}",
134        host,
135        port
136    );
137    let mut client = TimeoutStream::new(
138        hyper_util::rt::TokioIo::new(upgraded),
139    );
140    client.set_read_timeout(Some(TUNNEL_IDLE_TIMEOUT));
141    client.set_write_timeout(Some(TUNNEL_IDLE_TIMEOUT));
142    // `TimeoutStream` is not `Unpin`, and the copy needs `&mut` handles
143    // that outlive its internal futures, so pin them on the heap.
144    let mut client = Box::pin(client);
145
146    let mut target = TimeoutStream::new(target);
147    target.set_read_timeout(Some(TUNNEL_IDLE_TIMEOUT));
148    target.set_write_timeout(Some(TUNNEL_IDLE_TIMEOUT));
149    let mut target = Box::pin(target);
150
151    let buffer_size = crate::runtime_buffer_tuning::get().tunnel_buffer_size;
152    let (client_to_target, target_to_client) =
153        tokio::io::copy_bidirectional_with_sizes(
154            &mut client,
155            &mut target,
156            buffer_size,
157            buffer_size,
158        )
159            .await
160            .with_context(|| {
161                format!(
162                    "Tunnel {}:{} failed (idle timeout is {:?})",
163                    host, port, TUNNEL_IDLE_TIMEOUT
164                )
165            })?;
166    log_info!(
167        "[TUNNEL] Tunnel {}:{} closed normally after {:?}: client→target: {} bytes, target→client: {} bytes",
168        host,
169        port,
170        started.elapsed(),
171        client_to_target,
172        target_to_client
173    );
174    Ok(())
175}
